        % Code that executes after component creation
        function startupFcn(app)

            app.UIFigure.UserData = struct();
            app.UIFigure.UserData.x  = [];
            app.UIFigure.UserData.Fs = NaN;
            app.UIFigure.UserData.t  = [];
            app.UIFigure.UserData.y  = [];
        end

        % Button pushed function: LoadwavButton
        function LoadwavButtonPushed(app, event)
            [fn, fp] = uigetfile('*.wav','Select your WAV recording');
            file = fullfile(fp, fn);

            [x, Fs] = audioread(file);
            
            if size(x,2) > 1
                x = mean(x,2);
            end

            n = length(x); %sample sayısı
            Duration = n/Fs;
            t = (0:n-1)/Fs;
            
            Note = {"C4";"D4";"E4";"F4";"G4";"A4";"B4";"C5"};
            Measured_Hz = [261.5476 ; 293.2835 ; 329.3967 ; 350.1892 ; 391.7742 ; 439.9252 ; 494.6423 ; 524.1895];
            Theoretical_Hz = [261.63 ; 293.66 ; 329.63 ; 349.23 ; 392.00 ; 440.00 ; 493.88 ; 523.25];
            Error = [0.0315 ; 0.1282 ; 0.0708 ; 0.2747 ; 0.0576 ; 0.017	; 0.1543 ; 0.1796];
           

            app.UIFigure.UserData.x = x;
            app.UIFigure.UserData.Fs = Fs;
            app.UIFigure.UserData.t = t;

            app.FsTextArea.Value = num2str(Fs) + "Hz";
            app.DurationTextArea.Value = num2str(Duration) + "s";
            app.SamplesTextArea.Value = num2str(n);

            app.UITable.Data = table(Note,Measured_Hz,Theoretical_Hz,Error);

            plot(app.UIAxes,t,x,'r');
            grid(app.UIAxes,"on");

            X = fft(x);
            X = fftshift(X);
            Xn = length(X);
            X_f = (-Xn/2:Xn/2-1) * (Fs/Xn);

            plot(app.UIAxes_2,X_f,abs(X),'r');
            grid(app.UIAxes_2,"on");
            xlim(app.UIAxes_2,[-1500 1500]);

            

        end

        % Close request function: UIFigure
        function UIFigureCloseRequest(app, event)

        end

        % Button pushed function: ApplyFilterButton
        function ApplyFilterButtonPushed(app, event)
            if isempty(app.UIFigure.UserData.x) || isnan(app.UIFigure.UserData.Fs)
                uialert(app.UIFigure, 'First you have to load the .wav file.', 'Error');
                return;
            end
        
            x  = app.UIFigure.UserData.x;
            Fs = app.UIFigure.UserData.Fs;
            t  = app.UIFigure.UserData.t;
        
            % 2) Target notadan f0 bul
            noteNames = app.TargetDropDown.Value;
            order = str2num(app.OrderDropDown.Value);
            filter_type = app.FilterTypeDropDown.Value;

            if strcmpi(noteNames,'C4')
                f0 = 261.63;
            elseif strcmpi(noteNames,'D4')
                f0 = 293.66;
            elseif strcmpi(noteNames,'E4')
                f0 = 329.63;
            elseif strcmpi(noteNames,'F4')
                f0 = 349.23;
            elseif strcmpi(noteNames,'G4')
                f0 = 392.00;
            elseif strcmpi(noteNames,'A4')
                f0 = 440.00;
            elseif strcmpi(noteNames,'B4')
                f0 = 493.88;
            elseif strcmpi(noteNames,'C5')
                f0 = 523.25;
            end
            
            
            if strcmpi(filter_type,'Low_pass')
                Wn = f0/(Fs/2);
                [b,a] = butter(order, Wn, 'low');
                app.UIFigure.UserData.y = filtfilt(b,a,x);

    
            elseif strcmpi(filter_type,'High_pass')
                Wn = f0/(Fs/2);
                [b,a] = butter(order, Wn, 'high');
                app.UIFigure.UserData.y = filtfilt(b,a,x);
    
            else % Bandpass
                DeltaF = 15;                           % +/- bandwidth (Hz)

                bp = [f0-DeltaF f0+DeltaF];
                Wn = bp/(Fs/2);
                
                [A,B,C,D] = butter(order, Wn, 'bandpass');
                sos = ss2sos(A,B,C,D);
                y = sosfilt(sos,x);
                app.UIFigure.UserData.y = y;
            end

            plot(app.UIAxes_3,t,app.UIFigure.UserData.y,'r');
            grid(app.UIAxes_3,"on");
            

            Y = fft(y);
            Y = fftshift(Y);
            Y_n = length(Y);
            Y_f = (-Y_n/2:Y_n/2-1)*(Fs/Y_n);

            plot(app.UIAxes_4,Y_f,abs(Y)/max(abs(Y)),'r');
            grid(app.UIAxes_4,"on");
            xlim(app.UIAxes_4,[-1500 1500]);



        end

        % Button pushed function: PlayOriginalButton
        function PlayOriginalButtonPushed(app, event)
            if isempty(app.UIFigure.UserData.x) || isnan(app.UIFigure.UserData.Fs)
                uialert(app.UIFigure, 'First you have to load the .wav file.', 'Error');
                return;
            end

            x = app.UIFigure.UserData.x;
            Fs = app.UIFigure.UserData.Fs;
            soundsc(x,Fs);
        end

        % Button pushed function: PlayFilteredButton
        function PlayFilteredButtonPushed(app, event)
            if isempty(app.UIFigure.UserData.y) || isnan(app.UIFigure.UserData.Fs)
                uialert(app.UIFigure, 'First you have to apply the filter.', 'Error');
                return;
            end

            y = app.UIFigure.UserData.y;
            Fs = app.UIFigure.UserData.Fs;
            soundsc(y,Fs);
        end
    end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = GUI_app

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.UIFigure)

            % Execute the startup function
            runStartupFcn(app, @startupFcn)

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion
        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.UIFigure)
        end
    end
